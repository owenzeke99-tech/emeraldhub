local FileName = "EmeraldHub_KeyData.json"
local CacheDuration = 24 * 60 * 60 -- 24 giờ tính bằng giây

local function IsKeyValid()
    if readfile and isfile and isfile(FileName) then
        local success, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(FileName))
        end)
        if success and data and data.ExpiryTime then
            if os.time() < data.ExpiryTime then
                return true
            end
        end
    end
    return false
end

local function SaveKeyData()
    if writefile and game:GetService("HttpService") then
        pcall(function()
            local data = {
                ExpiryTime = os.time() + CacheDuration
            }
            writefile(FileName, game:GetService("HttpService"):JSONEncode(data))
        end)
    end
end

-- Hiệu ứng Hover Nút
local function AddHoverEffect(button, defaultColor, hoverColor)
    button.MouseEnter:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundColor3 = hoverColor}):Play()
    end)
    button.MouseLeave:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundColor3 = defaultColor}):Play()
    end)
end
AddHoverEffect(CheckBtn, Color_Emerald, Color_EmeraldHover)

-- Hàm Rung Lắc Mượt (Smooth Shake)
local function SmoothShake(guiObject)
    local originalX = -200
    local shakeOffsets = {-12, 12, -8, 8, -4, 4, 0} 
    
    for _, offset in ipairs(shakeOffsets) do
        local shakeTween = TweenService:Create(guiObject, TweenInfo.new(0.04, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {
            Position = UDim2.new(0.5, originalX + offset, 0.5, -110)
        })
        shakeTween:Play()
        task.wait(0.04)
    end
end
